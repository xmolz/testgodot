shader_type canvas_item;

uniform float dampness : hint_range(0.0, 1.0) = 0.2; // 0 is dry, 1 is soaked
uniform vec4 damp_tint : source_color = vec4(0.85, 0.9, 1.0, 1.0); // Slightly cool/blue tint

void fragment() {
    vec4 tex_color = texture(TEXTURE, UV);

    // Darken the colors and apply a very subtle cool tint to simulate wet fabric/skin
    vec3 wet_color = mix(tex_color.rgb, tex_color.rgb * damp_tint.rgb, dampness);
    wet_color = mix(wet_color, wet_color * 0.7, dampness); // 0.7 is the max darkening factor

    COLOR = vec4(wet_color, tex_color.a);
}

<file path="glass_water_drops.gdshader">
shader_type canvas_item;
uniform sampler2D screen_texture : hint_screen_texture, filter_linear_mipmap;
uniform float drop_speed : hint_range(0.1, 5.0) = 0.15;
uniform float drop_density : hint_range(1.0, 100.0) = 8.0;
uniform float distortion_strength : hint_range(0.0, 0.1) = 0.03;
uniform vec4 highlight_color : source_color = vec4(1.0, 1.0, 1.0, 0.3);

float random(vec2 uv) {
    return fract(sin(dot(uv.xy, vec2(12.9898,78.233))) * 43758.5453123);
}

void fragment() {
    vec2 uv = UV;
    vec2 screen_size = vec2(textureSize(screen_texture, 0));
    float aspect = screen_size.x / screen_size.y;
    vec2 drop_uv = vec2(uv.x * aspect, uv.y);

    float col = floor(drop_uv.x * drop_density);
    float random_col = random(vec2(col, 0.0));
    float t = TIME * drop_speed + random_col * 100.0;

    float slide_step = floor(t) + smoothstep(0.05, 0.3, fract(t));
    float speed_mult = mix(0.8, 1.2, random_col);
    drop_uv.y -= slide_step * speed_mult / drop_density;

    vec2 grid = fract(drop_uv * drop_density);
    vec2 cell = floor(drop_uv * drop_density);

    float cell_rand = random(cell);
    float drop_chance = step(0.4, cell_rand);
    vec2 gv = grid - vec2(0.5);

    float drop_size = mix(0.15, 0.3, cell_rand);
    float x_offset = mix(-0.2, 0.2, random(cell + vec2(1.0, 2.0)));
    vec2 drop_pos = vec2(x_offset, 0.25);

    vec2 delta = gv - drop_pos;
    vec2 head_delta = delta;
    head_delta.y *= 0.9;
    float head = smoothstep(drop_size, drop_size * 0.5, length(head_delta));

    float tail_taper = smoothstep(-0.5, drop_pos.y, gv.y);
    float tail_width = drop_size * 0.5 * tail_taper;
    float tail = smoothstep(tail_width, 0.0, abs(delta.x));

    tail *= smoothstep(-0.5, -0.2, gv.y);
    tail *= smoothstep(0.1, -0.1, delta.y);

    float water = max(head, tail) * drop_chance;

    vec2 static_uv = vec2(uv.x * aspect, uv.y);
    vec2 static_grid = fract(static_uv * drop_density * 2.5) - 0.5;
    vec2 static_cell = floor(static_uv * drop_density * 2.5);
    float static_rand = random(static_cell);
    float static_drop = smoothstep(0.2, 0.05, length(static_grid)) * step(0.7, static_rand);

    water = clamp(water + static_drop * 0.5, 0.0, 1.0);

    vec2 offset = vec2(0.0);
    if (water > 0.0) {
        vec2 normal = (gv - drop_pos) * 2.0;
        offset = normal * water * distortion_strength;
    }

    vec4 screen_color = texture(screen_texture, SCREEN_UV + offset);

    float highlight = smoothstep(drop_size * 0.8, drop_size * 0.4, length(head_delta - vec2(-0.02, -0.05))) * head;
    highlight = clamp(highlight, 0.0, 1.0) * drop_chance;

    vec3 final_color = screen_color.rgb;
    final_color = mix(final_color, highlight_color.rgb, water * highlight_color.a * 0.5);
    final_color = mix(final_color, vec3(1.0), highlight * 0.6);

    COLOR = vec4(final_color, screen_color.a);
}

<file path="shower_stream.gdshader">
shader_type canvas_item;

uniform sampler2D screen_texture : hint_screen_texture, filter_linear_mipmap;

uniform float stream_speed : hint_range(0.1, 10.0) = 3.0;
uniform float length_scale : hint_range(1.0, 50.0) = 15.0; // Stretches the noise along the line length
uniform float stream_scale : hint_range(1.0, 50.0) = 15.0; // Scales the droplet noise
uniform float distortion_strength : hint_range(0.0, 0.1) = 0.03;
uniform vec4 water_color : source_color = vec4(0.85, 0.95, 1.0, 0.3);
uniform float droplet_density : hint_range(0.1, 1.0) = 0.65;

// Basic noise generator
float random(vec2 uv) {
    return fract(sin(dot(uv.xy, vec2(12.9898,78.233))) * 43758.5453123);
}

float noise(vec2 uv) {
    vec2 i = floor(uv);
    vec2 f = fract(uv);
    f = f * f * (3.0 - 2.0 * f);
    float a = random(i);
    float b = random(i + vec2(1.0, 0.0));
    float c = random(i + vec2(0.0, 1.0));
    float d = random(i + vec2(1.0, 1.0));
    return mix(mix(a, b, f.x), mix(c, d, f.x), f.y);
}

void fragment() {
    // In a Line2D with STRETCH mode, UV.x is the length (0 to 1), UV.y is the width (0 to 1)
    vec2 uv = UV;

    // Scale UV.x so the noise doesn't stretch out weirdly on long lines
    vec2 scaled_uv = vec2(uv.x * length_scale, uv.y);

    // Create the scrolling motion along the X axis (down the line)
    float time = TIME * stream_speed;

    // Base UV for streaks (stretched heavily along the stream direction)
    vec2 streak_uv = vec2(scaled_uv.x - time, scaled_uv.y * 2.0) * stream_scale;
    streak_uv.x *= 0.15; // Stretch noise along the stream path

    // Multi-layered noise for complex, broken-up water spray
    float n1 = noise(streak_uv);
    float n2 = noise(streak_uv * 2.0 - vec2(time * 1.5, 0.0));
    float combined_noise = (n1 * 0.6 + n2 * 0.4);

    // Cut off noise to create separate droplets/streaks based on density
    float streaks = smoothstep(1.0 - droplet_density, 1.0, combined_noise);

    // Fade out the edges of the line so it isn't a harsh rectangle
    float edge_fade = smoothstep(0.0, 0.3, uv.y) * smoothstep(1.0, 0.7, uv.y);

    // Fade out the very beginning and end of the line
    float end_fade = smoothstep(0.0, 0.05, uv.x) * smoothstep(1.0, 0.9, uv.x);

    float water = streaks * edge_fade * end_fade;

    // Distort what is behind the water stream
    vec2 offset = vec2(0.0);
    if (water > 0.0) {
        // Calculate a fake normal for refraction
        offset = vec2(water * distortion_strength) * (uv.y - 0.5);
    }

    vec4 bg = texture(screen_texture, SCREEN_UV + offset);

    // Mix in the water color and add specular highlights
    vec3 final_color = mix(bg.rgb, water_color.rgb, water * water_color.a);
    final_color += vec3(water * 0.6); // Bright highlight for the drops

    // Soft anti-aliased alpha
    float alpha = smoothstep(0.01, 0.1, water);

    COLOR = vec4(final_color, alpha);
}

shader_type canvas_item;

uniform vec4 steam_color : source_color = vec4(0.9, 0.95, 1.0, 1.0);
uniform float fill_amount : hint_range(0.0, 1.0) = 0.4; // 0 is clear, 1 is a sauna
uniform float speed : hint_range(0.0, 2.0) = 0.3;
uniform float scale : hint_range(1.0, 10.0) = 3.0;

// Pseudo-random generator
float rand(vec2 n) {
    return fract(sin(dot(n, vec2(12.9898, 4.1414))) * 43758.5453);
}

// 2D Value Noise
float noise(vec2 p){
    vec2 ip = floor(p);
    vec2 u = fract(p);
    u = u*u*(3.0-2.0*u);
    float res = mix(
        mix(rand(ip),rand(ip+vec2(1.0,0.0)),u.x),
        mix(rand(ip+vec2(0.0,1.0)),rand(ip+vec2(1.0,1.0)),u.x),u.y);
    return res*res;
}

// Fractal Brownian Motion for cloud-like textures
float fbm(vec2 x) {
    float v = 0.0;
    float a = 0.5;
    vec2 shift = vec2(100.0);
    mat2 rot = mat2(vec2(cos(0.5), sin(0.5)), vec2(-sin(0.5), cos(0.50)));
    for (int i = 0; i < 4; ++i) {
        v += a * noise(x);
        x = rot * x * 2.0 + shift;
        a *= 0.5;
    }
    return v;
}

void fragment() {
    vec2 uv = UV * scale;
    // Steam drifts up and slightly right
    vec2 motion = vec2(TIME * speed * 0.2, -TIME * speed * 0.8);

    // Layer the FBM to create rolling, ethereal clouds
    float f = fbm(uv + motion);
    float f2 = fbm(uv + vec2(f) + motion * 1.2);
    float final_clouds = fbm(uv + vec2(f2) + motion * 1.5);

    // Smooth out the clouds and apply the overall room fill_amount
    float alpha = smoothstep(0.1, 0.9, final_clouds) * fill_amount;

    COLOR = vec4(steam_color.rgb, alpha * steam_color.a);
}

shader_type canvas_item;
uniform sampler2D DropsNoise : repeat_enable;
uniform float distortion_strength : hint_range(0.0, 5.0) = 0.5;
uniform float specular_intensity : hint_range(0.0, 3.0) = 1.0;
uniform float speed_multiplier : hint_range(0.1, 5.0) = 1.0;

float hash21(vec2 p) {
    uvec2 q = uvec2(ivec2(p)) * uvec2(1597334673U, 3812015801U);
    uint n = (q.x ^ q.y) * 1597334673U;
    return float(n) / float(0xffffffffU);
}

vec3 hash13(float p) {
   vec3 p3 = fract(vec3(p) * vec3(.1031,.11369,.13787));
   p3 += dot(p3, p3.yzx + 19.19);
   return fract(vec3((p3.x + p3.y)*p3.z, (p3.x+p3.z)*p3.y, (p3.y+p3.z)*p3.x));
}

float rainDrops(vec2 st, float time, float size, vec2 scale) {
    vec2 uv = st * size;
    uv.x *= scale.x / scale.y;
    vec2 gridUv = fract(uv) - .5;
    vec2 id = floor(uv);
    vec3 h = (hash13(id.x * 467.983 + id.y * 1294.387) - .5) * .8;
    vec2 dropUv = gridUv - h.xy;
    vec4 noise = texture(DropsNoise, id * .05);
    float drop = smoothstep(.25, 0., length(dropUv)) *
        max(0., 1. - fract(time * (noise.b + .1) * .2 + noise.g) * 2.);
    return drop;
}

vec2 wigglyDrops(vec2 st, float time, float size, vec2 scale) {
    vec2 wigglyDropAspect = vec2(2., 1.);
    vec2 uv = st * size * wigglyDropAspect;
    uv.x *= scale.x / scale.y;
    uv.y += time * .23;
    vec2 gridUv = fract(uv) - .5;
    vec2 id = floor(uv);
    float h = hash21(id);
    time += h * 2. * PI;
    float w = st.y * 10.;
    float dx = (h - .5) * .8;
    dx += (.3 - abs(dx)) * pow(sin(w), 2.) * sin(2. * w) *
        pow(cos(w), 3.) * 1.05;
    float dy = -sin(time + sin(time + sin(time) * .5)) * .45;
    dy -= (gridUv.x - dx) * (gridUv.x - dx);
    vec2 dropUv = (gridUv - vec2(dx, dy)) / wigglyDropAspect;
    float drop = smoothstep(.06, .0, length(dropUv));
    vec2 trailUv = (gridUv - vec2(dx, time * .23)) / wigglyDropAspect;
    trailUv.y = (fract((trailUv.y) * 8.) - .5) / 8.;
    float trailDrop = smoothstep(.03, .0, length(trailUv));
    trailDrop *= smoothstep(-.05, .05, dropUv.y) * smoothstep(.4, dy, gridUv.y) *
            (1.-step(.4, gridUv.y));
    float fogTrail = smoothstep(-.05, .05, dropUv.y) * smoothstep(.4, dy, gridUv.y) *
            smoothstep(.05, .01, abs(dropUv.x)) * (1.-step(.4, gridUv.y));
    return vec2(drop + trailDrop, fogTrail);
}

vec2 getDrops(vec2 st, float time, vec2 scale) {
    vec2 largeDrops = wigglyDrops(st, time * 2., 1.6, scale);
    vec2 mediumDrops = wigglyDrops(st + 2.65, (time + 1296.675) * 1.4, 2.5, scale);
    vec2 smallDrops = wigglyDrops(st - 1.67, time - 896.431, 3.6, scale);
    vec2 drops;
    drops.y = max(largeDrops.y, max(mediumDrops.y, smallDrops.y));
    drops.x = smoothstep(.4, 2., largeDrops.x + mediumDrops.x + smallDrops.x);
    return drops;
}

vec2 dropsUV(vec2 uv, float _time, vec2 scale) {
    // Invert Y so drops fall DOWN the screen, matching Godot's Y-down coordinate system
    vec2 st = vec2(uv.x, -uv.y);

    // Let time run forwards natively so the droplet physics play correctly
    float time = mod(_time, 7200.0);

    vec2 drops = getDrops(st, time, scale);
    vec2 dropsX = getDrops(st + vec2(0.001, 0.0), time, scale);

    // Sample Y in the negative direction to match the inverted Y-axis
    vec2 dropsY = getDrops(st + vec2(0.0, -0.001), time, scale);

    vec3 normal = vec3(dropsX.x - drops.x, dropsY.x - drops.x, 0.0);
    normal.z = sqrt(1.0 - normal.x * normal.x - normal.y * normal.y);
    normal = normalize(normal);

    return normal.xy;
}

void fragment() {
    vec2 scale = vec2(1.0, TEXTURE_PIXEL_SIZE.x / TEXTURE_PIXEL_SIZE.y);
    vec2 drop_normals = dropsUV(UV, TIME * speed_multiplier, scale);

    vec2 offset = drop_normals * (distortion_strength * 0.1);
    vec4 char_color = texture(TEXTURE, UV + offset);
    float base_alpha = texture(TEXTURE, UV).a;
    float shine = length(drop_normals) * specular_intensity;

    vec3 final_color = char_color.rgb;
    final_color += vec3(shine);
    COLOR = vec4(final_color, char_color.a * base_alpha);
}
